function finalizeMark(finalX, finalY) { isDone = true; popSound.play(); target.style.display = 'block'; userMark.style.left = `${finalX}px`; userMark.style.top = `${finalY}px`; userMark.style.display = 'block'; eye.style.display = 'none'; const xPct = (finalX / area.offsetWidth) * 100; const yPct = (finalY / area.offsetHeight) * 100; const tx = target.offsetLeft; const ty = target.offsetTop; const dist = Math.sqrt(Math.pow(finalX - tx, 2) + Math.pow(finalY - ty, 2)); let score = 0; // අලියාගේ ඇඟ ඇතුළතදැයි බලන පරීක්ෂාව if (isInside(xPct, yPct)) { // දුර අනුව ලකුණු ගණනය වේ (දුර වැඩි වන විට ලකුණු අඩු වේ, නමුත් 10% ට වඩා අඩු නොවේ) score = Math.max(10, 100 - (dist / 1.8)); // ඉතාම නිවැරදි නම් (දුර පරතරය 8px ට අඩු නම්) 99.8% ලකුණු ලැබේ if (dist < 8) score = 99.8; winSound.play(); if(score > 95) document.getElementById('status-txt').innerText = "EXCELLENT! 🏆"; else if(score > 60) document.getElementById('status-txt').innerText = "GOOD SHOT! 👍"; else document.getElementById('status-txt').innerText = "ON THE BODY! 🐘"; } else { score = 0; document.getElementById('status-txt').innerText = "OUTSIDE! 😅"; } document.getElementById('score-pct').innerText = score.toFixed(1) + "%"; resultCard.style.display = 'block'; topMsg.innerText = "GAME OVER"; }
